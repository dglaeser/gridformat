47 const std::vector<std::size_t> rank_points;

51 const std::size_t my_point_offset;
52
53 IOContext(int _my_rank,
54 int _num_ranks,
55 std::vector<std::size_t> _rank_cells,
56 std::vector<std::size_t> _rank_points)
57 : my_rank{_my_rank}
58 , num_ranks{_num_ranks}
59 , is_parallel{_num_ranks > 1}
60 , rank_cells{std::move(_rank_cells)}
61 , rank_points{std::move(_rank_points)}
62 , num_cells_total{_accumulate(rank_cells)}
63 , num_points_total{_accumulate(rank_points)}
64 , my_cell_offset{_accumulate_rank_offset(rank_cells)}
65 , my_point_offset{_accumulate_rank_offset(rank_points)} {
66 if (my_rank >= num_ranks)
67 throw ValueError(as_error("Given rank is not within communicator size"));
68 if (num_ranks != static_cast<int>(rank_cells.size()))
69 throw ValueError(as_error("Cells vector does not match communicator size"));
70 if (num_ranks != static_cast<int>(rank_points.size()))
71 throw ValueError(as_error("Points vector does not match communicator size"));
72 }

74 template<Concepts::Grid Grid, Concepts::Communicator Communicator>
75 static IOContext from(const Grid& grid, const Communicator& comm, int root_rank = 0) {
76 const int size = Parallel::size(comm);
77 const int rank = Parallel::rank(comm);
78 const std::size_t num_points = number_of_points(grid);
79 const std::size_t num_cells = number_of_cells(grid);
80 if (size == 1)
81 return IOContext{rank, size, std::vector{num_cells}, std::vector{num_points}};
82
83 const auto all_num_points = Parallel::gather(comm, num_points, root_rank);
84 const auto all_num_cells = Parallel::gather(comm, num_cells, root_rank);
85 const auto my_all_num_points = Parallel::broadcast(comm, all_num_points, root_rank);
86 const auto my_all_num_cells = Parallel::broadcast(comm, all_num_cells, root_rank);
87 return IOContext{rank, size, my_all_num_cells, my_all_num_points};
88 }

90 private:
91 std::size_t _accumulate(const std::vector<std::size_t>& in) const {
92 return std::accumulate(in.begin(), in.end(), std::size_t{0});
93 }
94
95 std::size_t _accumulate_rank_offset(const std::vector<std::size_t>& in) const {
96 if (in.size() <= static_cast<std::size_t>(my_rank))
97 throw ValueError("Rank-vector length must be equal to number of ranks");
98 return std::accumulate(in.begin(), std::next(in.begin(), my_rank), std::size_t{0});
99 }

118inline std::string extract_string(const Field& field) {
119 const auto serialization = field.serialized();
120 const auto characters = serialization.template as_span_of<const char>();
121 std::string result{characters.begin(), characters.end()};
122 while (!result.empty() && result.back() == '\0')
123 result.pop_back();
124 return result;
125}

172template<typename C>
173void check_version_compatibility(const HDF5::File<C>& file, const std::array<std::size_t, 2>& supported) {
174 if (file.has_attribute_at("/VTKHDF/Version"))

189template<Concepts::Communicator C, std::invocable<HDF5::File<>&> Write>
190void write_on_root(const std::string& filename, const C& comm, int root, Write&& write) {
191 Parallel::barrier(comm);
192 std::exception_ptr error;
193 if (Parallel::rank(comm) == root) {
194 try {
195 HDF5::File file{filename, HDF5::File<>::append};
196 write(file);
197 } catch (...) {
198 error = std::current_exception();
199 }
200 }
201 // the broadcast also keeps the other ranks from reopening the file before the root is done
202 if (Parallel::broadcast(comm, static_cast<int>(error != nullptr), root)) {
203 if (error)
204 std::rethrow_exception(error);
205 throw IOError("Writing to '" + filename + "' failed on rank " + std::to_string(root));
206 }
207}
