48struct IOContext {
49 const int my_rank;
50 const int num_ranks;
51 const bool is_parallel;
52 const std::vector<std::size_t> rank_cells;
53 const std::vector<std::size_t> rank_points;
54 const std::size_t num_cells_total;
55 const std::size_t num_points_total;
56 const std::size_t my_cell_offset;
57 const std::size_t my_point_offset;
58
59 IOContext(int _my_rank,
60 int _num_ranks,
61 std::vector<std::size_t> _rank_cells,
62 std::vector<std::size_t> _rank_points)
63 : my_rank{_my_rank}
64 , num_ranks{_num_ranks}
65 , is_parallel{_num_ranks > 1}
66 , rank_cells{std::move(_rank_cells)}
67 , rank_points{std::move(_rank_points)}
68 , num_cells_total{_accumulate(rank_cells)}
69 , num_points_total{_accumulate(rank_points)}
70 , my_cell_offset{_accumulate_rank_offset(rank_cells)}
71 , my_point_offset{_accumulate_rank_offset(rank_points)} {
72 if (my_rank >= num_ranks)
73 throw ValueError(as_error("Given rank is not within communicator size"));
74 if (num_ranks != static_cast<int>(rank_cells.size()))
75 throw ValueError(as_error("Cells vector does not match communicator size"));
76 if (num_ranks != static_cast<int>(rank_points.size()))
77 throw ValueError(as_error("Points vector does not match communicator size"));
78 }

80 template<Concepts::Grid Grid, Concepts::Communicator Communicator>
81 static IOContext from(const Grid& grid, const Communicator& comm, int root_rank = 0) {
82 const int size = Parallel::size(comm);
83 const int rank = Parallel::rank(comm);
84 const std::size_t num_points = number_of_points(grid);
85 const std::size_t num_cells = number_of_cells(grid);
86 if (size == 1)
87 return IOContext{rank, size, std::vector{num_cells}, std::vector{num_points}};
88
89 const auto all_num_points = Parallel::gather(comm, num_points, root_rank);
90 const auto all_num_cells = Parallel::gather(comm, num_cells, root_rank);
91 const auto my_all_num_points = Parallel::broadcast(comm, all_num_points, root_rank);
92 const auto my_all_num_cells = Parallel::broadcast(comm, all_num_cells, root_rank);
93 return IOContext{rank, size, my_all_num_cells, my_all_num_points};
94 }
95
96 private:
97 std::size_t _accumulate(const std::vector<std::size_t>& in) const {
98 return std::accumulate(in.begin(), in.end(), std::size_t{0});
99 }
100
101 std::size_t _accumulate_rank_offset(const std::vector<std::size_t>& in) const {
102 if (in.size() <= static_cast<std::size_t>(my_rank))
103 throw ValueError("Rank-vector length must be equal to number of ranks");
104 return std::accumulate(in.begin(), std::next(in.begin(), my_rank), std::size_t{0});
105 }
106};

210inline void write_field_data_step(HDF5::File<>& file, const std::string& name, FieldPtr field) {
211 const std::string path = "/VTKHDF/FieldData/" + name;
212 const auto offset = file.get_dimensions(path).value_or(std::vector<std::size_t>{0}).at(0);
213 std::array<std::size_t, 2> sizes;
214 if (is_string_field(*field)) {
215 const auto strings = extract_strings(*field);
216 sizes = {1, strings.size()};
217 file.write_strings(strings, path);
218 } else {
219 const auto tuples = as_field_data_tuples(field);
220 const auto layout = tuples->layout();
221 sizes = {layout.dimension() > 1 ? layout.extent(1) : 1, layout.extent(0)};
222 file.write(*tuples, path);
223 }
224 file.write(std::array{offset}, "/VTKHDF/Steps/FieldDataOffsets/" + name);
225 file.write(std::vector{sizes}, "/VTKHDF/Steps/FieldDataSizes/" + name);
226}

229inline void write_field_data_step_referencing(std::size_t step_index, HDF5::File<>& file, const std::string& name) {
230 const std::string offsets_path = "/VTKHDF/Steps/FieldDataOffsets/" + name;
231 const std::string sizes_path = "/VTKHDF/Steps/FieldDataSizes/" + name;
232 const auto offset = file.read_dataset_to<std::size_t>(
233 offsets_path, HDF5::Slice{.offset = {step_index}, .count = {1}}
234 );
235 const auto sizes = file.read_dataset_to<std::vector<std::size_t>>(
236 sizes_path, HDF5::Slice{.offset = {step_index, 0}, .count = {1, 2}}
237 );
238 file.write(std::array{offset}, offsets_path);
239 file.write(std::vector{std::array{sizes.at(0), sizes.at(1)}}, sizes_path);
240}

243template<typename C>
244FieldPtr read_field_data(const HDF5::File<C>& file, const std::string& name, std::optional<std::size_t> step) {
245 const std::string path = "VTKHDF/FieldData/" + name;
246 const auto precision = file.get_precision(path).value();
247 const bool is_string = precision.template is<char>();
248 if (!step)
249 return is_string ? make_field_ptr(RangeField{file.template read_dataset_to<MultiString>(path)})
250 : make_field_ptr(DataSetField{file, path});
251
252 // Without sizes (written by GridFormat <= 0.5), each step is a single row of the dataset
253 const auto offset = file.template read_dataset_to<std::size_t>(
254 "VTKHDF/Steps/FieldDataOffsets/" + name, HDF5::Slice{.offset = {*step}, .count = {1}}
255 );
256 const std::string sizes_path = "VTKHDF/Steps/FieldDataSizes/" + name;
257 const auto sizes = file.has_dataset_at(sizes_path) // (components, tuples)
258 ? std::optional{file.template read_dataset_to<std::vector<std::size_t>>(
259 sizes_path, HDF5::Slice{.offset = {*step, 0}, .count = {1, 2}}
260 )}
261 : std::nullopt;
262 if (is_string) {
263 const std::size_t count = sizes ? sizes->at(0)*sizes->at(1) : 1;
264 return make_field_ptr(RangeField{file.template read_dataset_to<MultiString>(
265 path, HDF5::Slice{.offset = {offset}, .count = {count}}
266 )});
267 }
268
269 const auto dims = file.get_dimensions(path).value();
270 std::vector<std::size_t> slice_offset(dims.size(), 0);
271 slice_offset.at(0) = offset;
272 auto count = dims;
273 count.at(0) = sizes ? sizes->at(1) : 1;
274 if (sizes && MDLayout{count}.number_of_entries() != sizes->at(0)*sizes->at(1))
275 throw SizeError("Field data '" + name + "' does not match its stated number of components");
276 MDLayout layout = sizes ? MDLayout{count} : MDLayout{count | std::views::drop(1)};
277 return make_field_ptr(DataSetField{
278 file,
279 std::move(layout),
280 precision,
281 [p=path, o=std::move(slice_offset), c=std::move(count)] (const HDF5::File<C>& f) {
282 return f.visit_dataset(p, [&] <typename F> (F&& field) {
283 return FlattenedField{make_field_ptr(std::move(field))}.serialized();
284 }, HDF5::Slice{.offset = o, .count = c});
285 }
286 });
287}

294template<Concepts::Communicator C, std::invocable<HDF5::File<>&> Write>
295void write_on_root(const std::string& filename, const C& comm, int root, Write&& write) {
296 Parallel::barrier(comm);
297 std::exception_ptr error;
298 if (Parallel::rank(comm) == root) {
299 try {
300 HDF5::File file{filename, HDF5::File<>::append};
301 write(file);
302 } catch (...) {
303 error = std::current_exception();
304 }
305 }
306 // the broadcast also keeps the other ranks from reopening the file before the root is done
307 if (Parallel::broadcast(comm, static_cast<int>(error != nullptr), root)) {
308 if (error)
309 std::rethrow_exception(error);
310 throw IOError("Writing to '" + filename + "' failed on rank " + std::to_string(root));
311 }
312}
