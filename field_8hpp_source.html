142 template<bool enable_resize, std::ranges::range R> requires(Concepts::Scalar<MDRangeValueType<R>>)
143 decltype(auto) _export_to(R&& output_range) const {
144 const auto my_layout = layout();
145
146 if constexpr (Concepts::ResizableMDRange<R> && enable_resize) {
147 const auto num_scalars = my_layout.number_of_entries();
148 const auto num_sub_scalars = get_md_layout<std::ranges::range_value_t<R>>().number_of_entries();
149 if (num_scalars%num_sub_scalars != 0)
150 throw TypeError(
151 "Cannot export the field into the given range type. "
152 "Number of entries in the field is not divisible by the "
153 "number of entries in the value_type of the provided range."
154 );
155
156 output_range.resize(
157 num_scalars/num_sub_scalars,
158 DefaultValue<std::ranges::range_value_t<R>>::get()
159 );
160 } else {
161 const auto output_range_layout = get_md_layout(output_range);
162 if (output_range_layout.number_of_entries() < my_layout.number_of_entries())
163 throw SizeError(
164 std::string{"Cannot fill the given range. Too few entries. "} +
165 "Number of field entries: '" + std::to_string(my_layout.number_of_entries()) + "'; " +
166 "Number of range entries: '" + std::to_string(output_range_layout.number_of_entries()) + "'"
167 );
168 }
169
170 std::size_t offset = 0;
171 visit_field_values([&] <typename T> (std::span<const T> data) {
172 _export_to(output_range, data, offset);
173 });
174
175 return std::forward<R>(output_range);
176 }

178 template<std::ranges::range R, Concepts::Scalar T>
179 void _export_to(R& range,
180 std::span<const T> data,
181 std::size_t& offset) const {
182 if constexpr (mdrange_dimension<R> > 1) {
183 std::ranges::for_each(range, [&] (std::ranges::range auto& sub_range) {
184 _export_to(sub_range, data, offset);
185 });
186 } else {
187 // Note: if the span is over a std::vector<bool>, the use of `std::ranges::copy` or similar broke here.
188 // It seems to work with more recent compilers and/or c++23, though. We use `std::copy` as a workaround.
189 auto converted_data = data | std::views::drop(offset)
190 | std::views::take(std::min(Ranges::size(range), data.size() - offset))
191 | std::views::transform([] (const T& value) {
192 return static_cast<std::ranges::range_value_t<R>>(value);
193 });
194 std::copy(
195 std::ranges::begin(converted_data),
196 std::ranges::end(converted_data),
197 std::ranges::begin(range)
198 );
199 offset += Ranges::size(converted_data);
200 }
201 }
